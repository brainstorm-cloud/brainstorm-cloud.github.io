// EVE'S STARTER — the remix core (canvas filter pipeline)
function remix(img, {bright=100, contrast=100, sat=100, filter=''} = {}) {
  const c = document.createElement('canvas');
  c.width = img.naturalWidth; c.height = img.naturalHeight;
  const x = c.getContext('2d');
  x.filter = `brightness(${bright}%) contrast(${contrast}%) saturate(${sat}%) ${filter}`;
  x.drawImage(img, 0, 0);
  return c; // export it, post it, or hand it to Eve
}
